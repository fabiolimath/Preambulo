\begin{tikzpicture}[scale=1]

% === Parâmetros ===
\pgfmathsetmacro{\r}{sqrt(2)}   % raio do círculo: x^2+y^2=2

  \begin{axis}[
      axis lines=middle,
      xlabel={$x$}, ylabel={$y$},
      axis line style={thick},
      axis equal image,                % mantém escala x=y para o círculo ficar circular
      set layers=axis on top,
      xmin=-1.3*\r, xmax=1.3*\r, ymin=-1.3*\r, ymax=1.3*\r,
      xtick={1,\r}, xticklabels={$1$,$\sqrt2$},
      ytick={-1}, yticklabels={$-1$},
      x tick label style={anchor=south west},
      y tick label style={anchor=south east},
      enlargelimits,
      clip=false,
      every axis x label/.style={at={(ticklabel* cs:0.99)}, anchor=west},
      every axis y label/.style={at={(ticklabel* cs:0.99)}, anchor=south},
    ]

    % --- Linhas Tracejadas ---
    \addplot[Tracejado] coordinates {(1,0) (1,-1)};
    \addplot[Tracejado] coordinates {(0,-1) (1,-1)};

    % --- Curvas ---
    \addplot[thick, blue, domain=0:360, samples=360]({\r*cos(x)},{\r*sin(x)})
      node[pos=0.13, above right, blue] {$x^2+y^2=2$};                        % Círculo
    \addplot[thick, red, domain=-1.2*\r:1.2*\r, samples=2] {-x}
      node[pos=0.95, below left, red] {$y=-x$};                              % Reta y=-x

    % --- Região sombreada: setor entre y=-x e y=0 ---
    \addplot[fill=gray!30, draw=none, opacity=0.6,
             domain=-45:0, samples=50, variable=\t]
      ({\r*cos(\t)}, {\r*sin(\t)})   % arco do círculo, de -45° a 0°
      -- (axis cs:0,0)               % volta à origem
      -- cycle;

    % --- Pontos de interseção ---
    \addplot[only marks, black!60, mark=*] coordinates {(0,0) (\r,0) (1,-1)};

  \end{axis}

\end{tikzpicture}
