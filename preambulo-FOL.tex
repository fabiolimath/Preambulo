\usepackage{calc}

% Define a new command with an optional argument for height
\newlength{\hght}
\newlength{\wdth}

% Seta para baixo atravessando a equação
% Se nada for passado, a seta é vermelha e aponta para zero
% No lugar do zero pode passar qq coisa como 1o argumento
% Ex.: \zer[1]{$equação$}
% O 2o argumento é a cor. Para usá-la, é necessário passar tbm o 1o argumento 
% Ex.: \zer[0][blue]{$equação$}
% O 3o argumento é o tamanho do texto no target da seta, por padrão \scriptsize
% O 4o argumento é para fazer ajuste fino na inclinação da seta
\newcommandx{\zer}[5][1=0, 2=red, 3=\scriptsize, 4=3ex]{%
    \setlength{\hght}{\heightof{$#5$}}\setlength{\wdth}{\widthof{$#5$}}%
    \makebox[0pt][l]{\tikz[baseline]{%
        % The vertical coordinates are swapped for a negative slope
        \draw[-latex, #2, thick](0,\hght+#4)--(\wdth,-#4) node[shift={(1.1mm,-.5mm)}]{#3 $#1$};%
    }}#5%
}

% Mesma coisa que a anterior, mas apontando para cima.
\newcommandx{\zerup}[5][1=0, 2=red, 3=\scriptsize, 4=3ex]{%
    \setlength{\hght}{\heightof{$#5$}}\setlength{\wdth}{\widthof{$#5$}}%
    \makebox[0pt][l]{\tikz[baseline]{%
        % #1 is the target value, #3 is the inclination
        \draw[-latex, #2, thick](0,-#4)--(\wdth,\hght+#4) node[shift={(1.1mm,.5mm)}]{#3 $#1$};%
    }}#5%
}
