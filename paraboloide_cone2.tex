\begin{tikzpicture}[scale=1]
  \begin{axis}[      
      xlabel={$y$}, ylabel={$z$},
      % xmin=-1.3, xmax=1.3,
      % ymin=-0.3, ymax=1.5,
      xtick={-1,0,1}, ytick={0,1},
      x tick label style={anchor=north east}, % rótulos abaixo e a esquerda do eixo X
      y tick label style={anchor=south east}, % rótulos acima e a esquerda do eixo y
      width=8cm, height=6cm,
      every axis x label/.style={at={(ticklabel* cs:0.97)}, anchor=north},
      every axis y label/.style={at={(ticklabel* cs:0.97)}, anchor=east},
    ]

    % --- Linhas Tracejadas ---
     \addplot[Tracejado] coordinates {(-1,0) (-1,1)};
     \addplot[Tracejado] coordinates {(-1,1) (1,1)};
     \addplot[Tracejado] coordinates {(1,0) (1,1)};

    % --- Curvas ---
    \addplot[name path=parabola, domain=-1.2:1.2, samples=100, thick, blue] {x^2}node[pos=0.7, below right, blue] {$z = y^2$};
    \addplot[name path=modulo, domain=-1.2:1.2, samples=100, thick, red] {abs(x)}node[pos=0.8, above left, red] {$z = |y|$};

    % --- Região sombreada entre as curvas ---
    \addplot[gray!30] fill between[of=parabola and modulo, soft clip={domain=-1:1}];

    % --- Pontos de interseção ---
    \addplot[only marks, black!60, mark=*] coordinates {(-1,1) (1,1) (0,0)};


  \end{axis}
  
\end{tikzpicture}
